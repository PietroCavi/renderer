    {
        threads
        thread_num
        void**arg;
        lock

basically all the threads get initialized and they will run a wrapper function which is structured like this:
    
    void* thread_func(void* arg){
        
        // extraction of function from arg
        
        while(1){
            wait();
            function(arg+offset);
        }

        return NULL;
    }

The init will return a structure containing the thread pointers, the number of threads, the lock and a portion of memory that will be used by the threads to
get their argument

the signature will look something like this:
    
    parallel_unit init_parallel_unit(size_t number_of_threads, void(*fnPtr)(void*));

The so called parallel_unit will be started with a simple function, start_parallel_unit(parallel_unit to_execute), it will look something 
like this (pseudo):

    void start_parallel_unit(parallel_unit to_execute){
        unlock(to_execute.lock);

        join_until_finished()

        return;
    }

will turn into this:
    
    static inline void scale(float x[24], float y[24], float z[24], float s){
        const __m256 V = _mm256_set1_ps(v);
        for(size_t i=0;i<24;i+=8){

            __m256 X   = _mm256_loadu_ps(x+i);
            __m256 Y   = _mm256_loadu_ps(y+i);
            __m256 Z   = _mm256_loadu_ps(z+i);

            _mm256_storeu_ps(x+i, _mm256_mul_ps(X, V));
            _mm256_storeu_ps(y+i, _mm256_mul_ps(Y, V));
            _mm256_storeu_ps(z+i, _mm256_mul_ps(Z, V));
        }
    }
